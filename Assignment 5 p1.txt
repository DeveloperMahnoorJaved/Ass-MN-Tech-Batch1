<!DOCTYPE html>
<html>
<head>

<body>

<th>Day</th>
<th>Date</th>

<th>Urdu</th>
<th>Math</th>
<th>Islamiat</th>
</tr>

<!-- Row 1 -->
<tr>

</tr>

<!-- Row 2 -->
<tr>
<td>Tuesday</td>
<td>02-05-2026</td>

</tr>

<!-- Row 3 -->
<tr>

</tr>

</table>

</body>
</html>

<tr> → table row
<th> → table heading (bold text)
<td> → table data (normal text)
border="1" → adds border to table
